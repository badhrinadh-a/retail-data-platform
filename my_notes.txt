################################################
docker exec -it retail-data-platform-kafka-1 bash
kafka-console-consumer \
  --bootstrap-server localhost:9092 \
  --topic orders \
  --from-beginning \
  --property print.key=true \
  --property key.separator=" : "

# command to see list of kafka topics
kafka-topics --bootstrap-server localhost:9092 --list

To see human readable MinIO buckets data locally
################################################
prereq:
1. download duckdb locally using command: curl https://install.duckdb.org | sh
2. make sure minio container (retail-data-platform-minio-1) is running

steps:
mc alias set myminio http://localhost:9000 minioadmin minioadmin

# list buckets and list bronze
mc ls myminio
mc ls myminio/bronze

mc ls --recursive myminio/bronze
mc ls myminio/bronze/events/topic=orders

mkdir -p orders_parquets
mc cp --recursive myminio/bronze/events/topic=orders ./orders_parquets    [copies all orders parquet files in bronze bucket to orders_parquets directory]
